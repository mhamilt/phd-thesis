%% Voltage Divider
\newcommand{\VDIV}[1][1]{%
\begin{circuitikz}[american, scale=#1, transform shape]
    \draw (0, 0)  node[left]{$V_{CC}$} to[short,o-] (0,0) to [R,l_=$R_{in}$] (0, -2);
    \draw -- (0,-2) node[npn, photo, anchor=C](Q){Q};
    \draw (Q.E) -- ++(0,0) node[ground] {};
    \draw (0,-2) to[short,*-o]  (0.5,-2) node[right]{$V_{out}$};
\end{circuitikz}
}

% Basic TIA (no VCE)
\newcommand{\TIABASIC}[1][1]{%
\begin{circuitikz}[american, scale=#1, transform shape]
	\draw (0, 0) node[op amp] (opamp) {};    
	
    \draw (opamp.-) to[short,-*] ++(-0.5, 0) coordinate(A);
	\draw (opamp.+) node[ground] {};
	\draw (opamp.out) to[short,*-o] ++(0.5, 0) coordinate(C) node[right]{$V_{out}$};

	\draw (A) |- ++(0.5,1) coordinate[yshift=1ex] (L1) to[R=$R_{f}$] ++(2,0) coordinate(Rl) -| (opamp.out);    
    \draw (A) |- ++(-0.5,-0) coordinate(Vout) node[npn, photo, anchor=C](Q){Q};
	\draw[lightgray] (Vout) to[short,*-] ++(0,0.2) to [R,l=$R_{in}$] ++(0, 1.5) to[short,-o] node[left]{$V_{CC}$} ++(0,0.5) coordinate(Vcc);
	\draw (Q.E) -- ++(0,0) node[ground] {};
\end{circuitikz}
}
